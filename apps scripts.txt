/**

function doGet(e) {

  if (action === 'getSchedule') {

      const slots = getFreeSlotsFromSheet(100); // Берем с запасом

        status: 'success',
        data: slots
      };

    } catch (err) {
      const errorResult = { status: 'error', message: err.toString() };
      return ContentService.createTextOutput(JSON.stringify(errorResult))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // 2. Если просто зашли проверить статус (браузер или телеграм пинг)
  return ContentService.createTextOutput("Bot is running. Webhook active.")
         .setMimeType(ContentService.MimeType.TEXT);
}

/**
 * Обработка POST запросов (от Telegram бота)
 */
function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput("Invalid request");
    }
    
    const json = JSON.parse(e.postData.contents);
    return handleLogic(json);
  } catch (err) {
    Logger.log("Ошибка в doPost: " + err.toString());
    return ContentService.createTextOutput("OK"); 
  }
}

 * Логика обработки событий от Telegram

    return ContentService.createTextOutput("OK");

  return ContentService.createTextOutput("OK");

/**
 * Обработка кнопок
 */
function handleCallback(callbackData) {
  const callbackId = callbackData.id;
  const chatId = callbackData.message.chat.id;
  const messageId = callbackData.message.message_id;
  
  let actionData;
  try {
    actionData = JSON.parse(callbackData.data);
  } catch (e) {
    Logger.log("Ошибка парсинга callback: " + e);
    return;
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const row = actionData.row;
  
  if (!row || !sheet) return;

  let newStatus = '';
  let messageText = '';

  if (actionData.action === 'accept') {
    newStatus = 'Занято';
    messageText = `✅ <b>ЗАПИСЬ ПОДТВЕРЖДЕНА!</b>\nКлиент: ${actionData.clientName}\nДата: ${actionData.date}\nВремя: ${actionData.time}`;
  } else if (actionData.action === 'reject') {
    newStatus = 'Отклонено';
    messageText = `❌ <b>ЗАПИСЬ ОТКЛОНЕНА</b>\nКлиент: ${actionData.clientName}`;
  }

  if (newStatus) {
    try {
      sheet.getRange(row, 4).setValue(newStatus);
      
      const url = `https://api.telegram.org/bot${BOT_TOKEN}/editMessageText`;
      const payload = {
        chat_id: chatId,
        message_id: messageId,
        text: messageText,
        parse_mode: 'HTML',
        reply_markup: JSON.stringify({ inline_keyboard: [] })
      };
      
      UrlFetchApp.fetch(url, {
        method: 'post',
        contentType: 'application/json',
        payload: JSON.stringify(payload),
        muteHttpExceptions: true
      });
    } catch (e) {
      Logger.log("Ошибка обновления: " + e);
    }
  }
}

/**
 * Получение слотов из таблицы
 */
function getFreeSlotsFromSheet(limit) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) return [];

  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];

  const now = new Date();
  const freeSlots = [];
  
  const COL_DATE = 0;
  const COL_TIME = 1;
  const COL_STATUS = 3;

  for (let i = 1; i < data.length; i++) {
    const rowDateVal = data[i][COL_DATE];
    const rowTimeVal = data[i][COL_TIME];
    const status = data[i][COL_STATUS];

    if (['Занято', 'Отклонено', 'Отмена'].includes(status)) continue;
    if (!(rowDateVal instanceof Date) || isNaN(rowDateVal)) continue;

    const slotDate = new Date(rowDateVal);
    
    if (typeof rowTimeVal === 'string' && rowTimeVal.includes(':')) {
      const parts = rowTimeVal.split(':');
      slotDate.setHours(parseInt(parts[0], 10), parseInt(parts[1], 10), 0);
    } else {
      continue;
    }

    if (slotDate > now) {
      freeSlots.push({
        date: Utilities.formatDate(rowDateVal, Session.getScriptTimeZone(), "dd.MM.yyyy"),
        time: rowTimeVal,
        timestamp: slotDate.getTime()
      });
      if (freeSlots.length >= limit) break;
    }
  }

  freeSlots.sort((a, b) => a.timestamp - b.timestamp);
  return freeSlots;
}

/**
 * Отправка сообщений в Telegram
 */
function sendTelegramMessage(chatId, text, replyMarkup = null) {
  const url = `https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`;
  const payload = {
    chat_id: chatId,
    text: text,
    parse_mode: 'HTML'
  };
  
  if (replyMarkup) {
    payload.reply_markup = JSON.stringify(replyMarkup);
  }

  try {
    UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });
  } catch (e) {
    Logger.log("Ошибка отправки: " + e);
  }
}

/**
 * Триггер сортировки (работает при ручном редактировании)
 */
function onEdit(e) {
  if (!e || !e.range) return;
  const sheet = e.range.getSheet();
  if (sheet.getName() === SHEET_NAME) {
    try {
      sheet.sort([{column: 1, ascending: true}, {column: 2, ascending: true}]);
    } catch (err) { /* игнор */ }
  }
}
